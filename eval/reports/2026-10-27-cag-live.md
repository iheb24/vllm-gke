```bash
# Port-forward the Envoy gateway (model traffic)
export ENVOY_SERVICE=$(kubectl get svc -n envoy-gateway-system \
  --selector=gateway.envoyproxy.io/owning-gateway-namespace=vllm,gateway.envoyproxy.io/owning-gateway-name=semantic-router \
  -o jsonpath='{.items[0].metadata.name}')

# Port-forward the vLLM pod's /metrics (counter scraping)
kubectl port-forward -n vllm deploy/vllm-server 8000:8000 &

# Env (never commit the key)
export GATEWAY_API_KEY=$(kubectl get secret vllm-api-key -n vllm -o jsonpath='{.data.key}' | base64 -d)
export GATEWAY_URL=http://localhost:8081/v1

# Start the demo WITHOUT the fake model (live mode)
.venv-showcases/bin/uvicorn app:app --port 8092 --app-dir showcases/cag &

# Smoke check, then explicit warm-up (primes the prefix cache)
curl -s http://localhost:8092/healthz
curl -s -X POST http://localhost:8092/warmup

# Benchmark: readiness wait, cold-prefix prime (measured alone), warm loop
# of >=10 streaming requests, counter deltas, VERIFIED/NOT VERIFIED verdict.
.venv-showcases/bin/python showcases/cag/benchmark.py \
  --app-url http://localhost:8092 \
  --vllm-metrics-url http://localhost:8000/metrics \
  --questions eval/showcases/cag_questions.yaml \
  --n-warm 10 \
  --report eval/reports/$(date +%F)-cag-live-benchmark.json

# Quality eval against the live model
.venv-showcases/bin/python eval/run_cag_eval.py \
  --url http://localhost:8092 \
  --questions eval/showcases/cag_questions.yaml \
  --report eval/reports/$(date +%F)-cag-live.md \
  --note "Live run: $GATEWAY_MODEL via Envoy gateway; image digest <fill in>; vLLM <version>; prefix caching <confirmed>; counters <names used>."

# Optional pod-restart lifecycle test (explicitly gated; deletes the vLLM
# pod to prove the cache is process-local and the first request re-primes):
AG_ALLOW_POD_RESTART=1 .venv-showcases/bin/python showcases/cag/benchmark.py \
  --app-url http://localhost:8092 \
  --vllm-metrics-url http://localhost:8000/metrics \
  --questions eval/showcases/cag_questions.yaml \
  --n-warm 5 --restart-test \
  --report eval/reports/$(date +%F)-cag-live-restart.json
```
